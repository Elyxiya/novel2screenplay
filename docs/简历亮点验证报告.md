class Semaphore {
  private current = 0;
  private queue: Array<() => void> = [];
  constructor(private max: number) {}
  async acquire(): Promise<void> {
    if (this.current < this.max) { this.current++; return; }
    return new Promise(r => this.queue.push(r));
  }
  release(): void { const n = this.queue.shift(); if (n) n(); else this.current--; }
}

#### 四层降级 JSON 解析

[safe-json.ts:9-62](src/lib/utils/safe-json.ts#L9-L62) — `safeJsonParse` 函数，严格按四层策略执行：

```typescript
export function safeJsonParse(text: string): unknown {
  // Strategy 1: Direct parse (fastest)
  try { return JSON.parse(text); } catch { /* fall through */ }

  // Strategy 2: Strip markdown code fences
  const stripped = text
    .replace(/^```json\s*/i, '')
    .replace(/^```\s*/i, '')
    .replace(/\s*```$/, '')
    .trim();
  try { return JSON.parse(stripped); } catch { /* fall through */ }

  // Strategy 3: Find first JSON object or array using bracket matching
  const firstObj = text.indexOf('{');
  const firstArr = text.indexOf('[');
  const start = firstObj === -1 ? firstArr : firstArr === -1 ? firstObj : Math.min(firstObj, firstArr);
  if (start !== -1) {
    let depth = 0;
    for (let i = start; i < text.length; i++) {
      if (text[i] === '{' || text[i] === '[') depth++;
      else if (text[i] === '}' || text[i] === ']') { depth--; if (depth === 0) { end = i; break; } }
    }
    try { return JSON.parse(text.slice(start, end + 1)); } catch { /* fall through */ }
  }

  // Strategy 4: Strip trailing commas and common garbage
  const cleaned = stripped
    .replace(/,(\s*[}\]])/g, '$1')
    .replace(/([{,]\s*)(\w+)\s*:/g, '$1"$2":')
    .replace(/:\s*'([^']*)'/g, ': "$1"')
    .replace(/[\x00-\x1F]/g, '')
    .trim();
  try { return JSON.parse(cleaned); } catch { return { _parseError: true, originalText: text }; }
}
```

#### 防幻觉 Prompt

[convert-scene.ts:5-72](src/lib/llm/prompts/convert-scene.ts#L5-L72) — Phase 3 系统 Prompt 包含 8 条防幻觉规则：

- **规则 1**：action 内容必须可见可听 — 禁止心理活动、背景知识、原文直接复制
- **规则 2**：对话归属规则 — 原文无明确说话人的对白归属"围观者"，不得归属具名角色
- **规则 3**：对话必须来自原文 — 只保留原文有明确引号的对白
- **规则 4**：每个 content 块必须有 sourceRefs
- **规则 8**：严格基于原文 — 原文未出现的角色、对话、事件不要添加

具体约束示例（Prompt 节选）：

```
1. 【action 内容必须可见可听】
   ✅ 正确：人物的动作、表情、场景变化
   ❌ 错误：心理活动（"他想..."、"心中浮现..."）、背景介绍（"大陆名为..."）

2. 【对话归属规则 — 最关键】
   - 原文引号内无明确说话人 → characterId 设为 "围观者"
   - 绝对禁止：把围观者的嘲讽对白归属给具名主角

3. 【对话必须来自原文】
   只保留原文中有明确引号的对白，不要自己编造或补全对话。
```

#### Levenshtein 距离角色名去重

[Phase4Merger.ts:10-48](src/lib/pipeline/Phase4Merger.ts#L10-L48) — 完整 Levenshtein DP 实现：

```typescript
function levenshtein(a: string, b: string): number {
  const m = a.length; const n = b.length;
  const dp: number[][] = Array.from({ length: m + 1 }, () => Array(n + 1).fill(0));
  for (let i = 0; i <= m; i++) dp[i][0] = i;
  for (let j = 0; j <= n; j++) dp[0][j] = j;
  for (let i = 1; i <= m; i++) {
    for (let j = 1; j <= n; j++) {
      if (a[i - 1] === b[j - 1]) dp[i][j] = dp[i - 1][j - 1];
      else dp[i][j] = Math.min(dp[i - 1][j - 1], dp[i - 1][j], dp[i][j - 1]) + 1;
    }
  }
  return dp[m][n];
}

function isSimilarName(a: string, b: string): boolean {
  if (a === b) return true;
  const normA = a.replace(/[^一-鿿\w]/g, '');
  const normB = b.replace(/[^一-鿿\w]/g, '');
  if (normA === normB) return true;

  // 包含关系：短名 ≥2 字且被长名包含
  const shorter = normA.length <= normB.length ? normA : normB;
  const longer = normA.length > normB.length ? normA : normB;
  if (shorter.length >= 2 && longer.includes(shorter)) return true;

  // Levenshtein 距离阈值 0.3
  const dist = levenshtein(normA, normB);
  return dist / Math.max(normA.length, normB.length, 1) < 0.3;
}
```

```typescript
const VALID_TIME_OF_DAY = new Set([
  'dawn', 'morning', 'afternoon', 'dusk', 'night', 'late-night', 'unknown'
]);

const TIME_OF_DAY_FALLBACKS: Record<string, string> = {
  day: 'afternoon', daytime: 'afternoon', '日': 'afternoon', '白天': 'afternoon',
  '中午': 'afternoon', noon: 'afternoon', '早晨': 'morning', '早上': 'morning',
  '上午': 'morning', '清晨': 'dawn', '黎明': 'dawn', '黄昏': 'dusk', '傍晚': 'dusk',
  '晚上': 'night', '夜晚': 'night', '深夜': 'late-night', midnight: 'late-night',
};

function normalizeTimeOfDay(value: unknown): string {
  const normalized = String(value ?? '').toLowerCase().trim();
  if (VALID_TIME_OF_DAY.has(normalized)) return normalized;
  if (TIME_OF_DAY_FALLBACKS[normalized]) return TIME_OF_DAY_FALLBACKS[normalized];
  return 'unknown';
}
```
